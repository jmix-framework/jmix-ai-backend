# Jmix @Internal API pages excluded from the javaapi corpora.
#
# Published Javadoc HTML does not render the @Internal annotation, so this filter
# cannot be derived from the pages themselves; the list is generated from the
# framework sources: class-level @Internal types and packages whose package-info
# is annotated with @Internal.
#
# Union of jmix-framework/jmix tags v2.8.2 and v3.0.0. Regenerate from a jmix clone:
#   for TAG in v2.8.2 v3.0.0; do
#     git grep -l '^@Internal' "$TAG" -- '*/src/main/java/io/jmix/*.java' \
#       | grep -v package-info | sed -E "s#^$TAG:.*src/main/java/(io/jmix/.*)\.java#\1.html#"
#     git grep -l '@Internal' "$TAG" -- '*/src/main/java/io/jmix/*package-info.java' \
#       | sed -E "s#^$TAG:.*src/main/java/(io/jmix/.*)/package-info\.java#\1/#"
#   done | sort -u
#
# Entries ending with "/" are @Internal packages: every class page directly in that
# directory is excluded. Other entries are class pages; nested-class pages of an
# excluded class (Outer.Inner.html) are excluded together with it.
#
# Exceptions live in internal-whitelist.txt: @Internal pages the official documentation
# teaches to use stay in the corpora and override both class and package entries here.
